{%- comment -%} Shows the artwork photograph when the file is actually in the repository, and a generated gradient panel with the piece's initials when it is not — so a listing added in the CMS before its photo is uploaded still looks deliberate instead of broken. Usage: {% include art-image.html src=item.image alt="..." title=item.title %} {%- endcomment -%} {%- assign art_file = site.static_files | where: "path", include.src | first -%} {%- if include.src and art_file -%} {%- else -%} {%- assign art_words = include.title | default: "Artwork" | split: " " -%} {%- capture art_initials %}{{ art_words[0] | slice: 0, 1 }}{% if art_words[1] %}{{ art_words[1] | slice: 0, 1 }}{% endif %}{% endcapture -%} {{ art_initials | upcase }} {%- endif -%}
